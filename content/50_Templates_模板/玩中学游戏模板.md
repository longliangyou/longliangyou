        allow="fullscreen" allowfullscreen></iframe>

<script>
(() => {
  if (window.__iframeFullscreenBridge) return;
  window.__iframeFullscreenBridge = true;

  const fullscreenElement = () => document.fullscreenElement ||
    document.webkitFullscreenElement ||
    document.mozFullScreenElement ||
    document.msFullscreenElement;

  const requestFullscreen = el => {
    const fn = el.requestFullscreen ||
      el.webkitRequestFullscreen ||
      el.mozRequestFullScreen ||
      el.msRequestFullscreen;
    return fn ? fn.call(el) : Promise.reject(new Error('Fullscreen API unavailable'));
  };

  const exitFullscreen = () => {
    const fn = document.exitFullscreen ||
      document.webkitExitFullscreen ||
      document.mozCancelFullScreen ||
      document.msExitFullscreen;
    return fn ? fn.call(document) : Promise.reject(new Error('Fullscreen API unavailable'));
  };

  const findFrame = source => Array.from(document.querySelectorAll('iframe'))
    .find(frame => frame.contentWindow === source);

  const sendState = () => {
    const active = fullscreenElement();
    document.querySelectorAll('iframe').forEach(frame => {
      frame.contentWindow?.postMessage({
        type: 'iframe-fullscreen-state',
        active: active === frame
      }, '*');
    });
  };

  window.addEventListener('message', event => {
    const type = event.data && event.data.type;
    if (type !== 'iframe-fullscreen-request' && type !== 'iframe-fullscreen-exit') return;

    if (type === 'iframe-fullscreen-exit') {
      if (fullscreenElement()) Promise.resolve(exitFullscreen()).finally(sendState);
      return;
    }

    const frame = findFrame(event.source);
    if (!frame) return;
    Promise.resolve(requestFullscreen(frame)).finally(sendState);
  });

  ['fullscreenchange', 'webkitfullscreenchange', 'mozfullscreenchange', 'MSFullscreenChange']
    .forEach(eventName => document.addEventListener(eventName, sendState));
})();
</script>
